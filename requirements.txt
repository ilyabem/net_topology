# Net Topology - Python dependencies
# Install: pip install -r requirements.txt
# Python 3.10+ (tested on 3.10 - 3.14)
#
# Keep this file ASCII-only: older pip versions on Windows read it in the
# locale code page (cp1252/cp1251) and fail on non-ASCII characters.

# Topology graph (layout is pure Python, numpy is not required)
networkx>=3.0

# GUI (QtSvg and QtPrintSupport are included: icons, PDF export)
PyQt5>=5.15.0

# Port scanning (requires the nmap program: https://nmap.org/download.html)
python-nmap>=0.7.1

# L2 ARP scanning (requires root / Administrator + Npcap on Windows)
scapy>=2.5.0

# SNMP (optional). pysnmp 4.4.12 works only on Python <= 3.11 (asyncore was
# removed in 3.12). On newer Python install the net-snmp "snmpget" utility,
# it is picked up automatically.
pysnmp==4.4.12; python_version < "3.12"
# pysnmp 4.4.12 imports pyasn1.compat.octets, removed in pyasn1 0.6.1
pyasn1<0.6.1; python_version < "3.12"

# MAC vendor lookup fallback (when the OUI database is not downloaded yet)
mac-vendor-lookup>=0.1.12

# Tests
pytest>=7.0
